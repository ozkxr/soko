# 1. CONFIG DE PIN
CONFIG PIN 5 OUT
CONFIG PIN 4 IN
CONFIG PIN 25 ANALOG_OUT
CONFIG PIN 34 ANALOG_IN

# 2. ASIGNACIÓN DIGITAL
SET PIN 5 = ON
SLEEP 1000
SET PIN 5 = OFF

# 2.1 ASIGNACIÓN ANALÓGICA
SET PIN 25 = 128
SET PIN 25 = 3.3

# 3. BUCLE WHILE
WHILE (PIN 4) DO
    SET PIN 5 = ON
    SLEEP 500
    SET PIN 5 = OFF
    SLEEP 500
ENDWHILE

# 3.1 WHILE CON PIN ANALÓGICO
WHILE (PIN 34 < 2000) DO
    SET PIN 25 = 512
    SLEEP 100
ENDWHILE

# 4. BUCLE REPEAT
# Repite un bloque de código un número de veces
REPEAT 30 DO
    SET PIN 5 = ON
    SLEEP 100
    SET PIN 5 = OFF
    SLEEP 100
ENDREPEAT

# 5. CONDICIONAL IF
IF (PIN 4) THEN
    SET PIN 5 = ON
ENDIF

# 5.1 CONDICIONAL IF / ELSE
IF (!PIN 5) THEN
    SET PIN 5 = OFF
ELSE
    SET PIN 5 = ON
ENDIF

# 5.2 CONDICIONAL CON COMPARADORES ANALÓGICOS
IF (PIN 34 >= 3000) THEN
    SET PIN 5 = ON
ELSE
    SET PIN 5 = OFF
ENDIF

IF (PIN 34 <= 100) THEN
    SET PIN 25 = 0.0
ENDIF

SLEEP 4000
